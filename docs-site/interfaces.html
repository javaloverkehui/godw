2. 连接器

package connector

type Capability struct {
    MergeUpsert          bool // 支持 MERGE / ON CONFLICT
    InsertOverwrite      bool // 支持 INSERT OVERWRITE PARTITION
    ReplacePartition     bool // 支持 ALTER TABLE ... REPLACE PARTITION（ClickHouse）
    TransactionalDDL     bool
    MaterializedView     bool
    WindowFunction       bool
    ApproxCountDistinct  bool
    JSONType             bool
    BulkLoad             bool
}

type TableMeta struct {
    Schema      string        `json:"schema"`
    Name        string        `json:"name"`
    Type        string        `json:"type"`
    Comment     string        `json:"comment"`
    RowCount    int64         `json:"rowCount"`
    SizeBytes   int64         `json:"sizeBytes"`
    Columns     []ColumnMeta  `json:"columns"`
    PartitionBy []string      `json:"partitionBy"`
}

type ColumnMeta struct {
    Name     string  `json:"name"`
    Type     string  `json:"type"`      // 引擎原始类型
    Nullable bool    `json:"nullable"`
    Default  *string `json:"default"`
    Comment  string  `json:"comment"`
    Ordinal  int     `json:"ordinal"`
}

// Connector 是数据源的唯一入口。实现必须并发安全（会被连接池复用）。
type Connector interface {
    Type() string
    Capabilities() Capability
    Dialect() dialect.Dialect

    Test(ctx context.Context) error
    ListSchemas(ctx context.Context) ([]string, error)
    ListTables(ctx context.Context, schema string) ([]TableMeta, error)
    DescribeTable(ctx context.Context, schema, table string) (*TableMeta, error)

    // Query 返回游标式结果，调用方负责 Close。必须支持 ctx 取消。
    Query(ctx context.Context, sql string, args ...any) (Rows, error)
    Exec(ctx context.Context, sql string, args ...any) (Result, error)

    // Stream 用于大结果集导出，按 batchSize 分批回调。
    Stream(ctx context.Context, sql string, batchSize int, fn func(Rows) error) error

    Close() error
}

type Rows interface {
    Columns() []ColumnMeta
    Next() bool
    Scan(dest ...any) error
    Err() error
    Close() error
}

type Result struct {
    RowsAffected int64
    Duration     time.Duration
}

// 注册表：各连接器在 init() 中注册，核心代码不感知具体实现。
func Register(dsType string, f Factory)
func Open(ctx context.Context, ds *meta.Datasource, secret *Secret) (Connector, error)

连接池：按 datasource_id 复用 Connector，池上限来自数据源配置，空闲 10 分钟回收。禁止每个任务新建连接。

3. 方言

type Dialect interface {
    Name() string
    QuoteIdent(string) string
    Placeholder(i int) string                 // $1 / ? / ?
    LimitOffset(limit, offset int64) string
    DateTrunc(granularity string, expr string) string
    DateAdd(expr string, n int, unit string) string
    CastType(logical string) string           // 逻辑类型 → 引擎类型
    Func(name string, args ...string) string  // 函数名映射，如 ifnull→coalesce
    SupportsFeature(f Feature) bool
}

为什么不用字符串模板拼方言：函数名差异（DATE_SUB vs dateSub vs date_sub）、分页差异（LIMIT n,m vs LIMIT n OFFSET m）、类型差异（TEXT vs String vs Nullable(String)）必须集中管理，否则每加一个引擎就是一次全量回归。

type ModelSpec struct {
    Kind          string       `yaml:"kind"`          // Model

type ColumnSpec struct {
    Name     string `yaml:"name"`
    Type     string `yaml:"type"`      // 逻辑类型：string|int64|decimal(18,4)|date|timestamp|bool|json
    Nullable bool   `yaml:"nullable"`
    Default  string `yaml:"default"`

    PII      string `yaml:"pii"`       // none|low|medium|high
}

// Compile 纯函数：无 IO、无全局状态，便于表驱动测试。
func Compile(spec ModelSpec, d dialect.Dialect) (ddl string, warnings []string, err error)

// Diff 比较两个版本的列差异，判定是否为破坏性变更。
type Change struct {
    Kind   string // add_column|drop_column|widen_type|narrow_type|rename|change_partition
    Column string
    From   string
    To     string
    Breaking bool
}
func Diff(oldSpec, newSpec ModelSpec) []Change

分层校验规则（编译器强制）

层 命名 允许引用 必须包含 分区
ODS ods_ 数据源表 dt、_godw_src、_godw_load_at dt
DWD dwd_ ODS / DWD 业务主键、dt dt
DWS dws_ DWD / DWS 统计粒度字段、dt dt
ADS ads_ DWD / DWS / ADS 面向场景的宽表字段 dt 可选

违反任一条 → 编译失败并给出可定位的错误（modelc.ErrLayerViolation{Model, Ref, Rule}）。

5. 管道 DSL

kind: Pipeline
name: daily_order_pipeline
description: 订单主题日链路
schedule:
  cron: "0 2 * * *"
  timezone: Asia/Shanghai
  startDate: "2024-01-01"
  maxConcurrency: 2
params:
  - { name: region, type: string, default: ALL }
resources:
  group: etl
  timeout: 2h
  retry: { max: 3, backoff: exponential, interval: 30s }
nodes:
  - key: load_ods
    type: source
    datasource: mysql_main
    sql: |
      SELECT * FROM orders WHERE DATE(updated_at) = '${sys.bizdate}'
    write:
      model: ods_order
      strategy: overwrite_partition
      partition: "${sys.bizdate}"

  - key: clean
    type: sql
    dependsOn: [load_ods]
    sql: |
      SELECT order_id, sku_id, CAST(amount AS DECIMAL(18,4)) AS amount, dt
      FROM {{ ref('ods_order') }}
      WHERE dt = '${sys.bizdate}' AND amount > 0
    write:
      model: dwd_order_detail
      strategy: merge
      keys: [order_id, sku_id]

  - key: check
    type: quality
    dependsOn: [clean]
    rules: [dwd_order_detail.amount_not_null]

  - key: agg
    type: sql
    dependsOn: [check]
    sql: |
      SELECT dt, shop_id, COUNT(DISTINCT order_id) AS order_cnt,
             SUM(amount) AS gmv
      FROM {{ ref('dwd_order_detail') }}
      WHERE dt = '${sys.bizdate}'
      GROUP BY dt, shop_id
    write:
      model: dws_order_shop_1d
      strategy: overwrite_partition
      partition: "${sys.bizdate}"

5.1 节点类型

type 作用 关键字段
source 从数据源抽取并写入 ODS datasource, sql/table, write
sql 任意 SQL 转换 sql, write
quality 质量门禁 rules, onFail: block\|warn
branch 条件分支 condition, then, else
subpipeline 嵌套子管道 pipeline, params
shell 执行外部脚本 command, timeout
http 调用外部接口 method, url, body
sensor 等待外部条件 type: file\|table\|http, interval, deadline

5.2 参数命名空间

命名空间 示例 说明
${sys.*} ${sys.bizdate} ${sys.bizdate-1d} ${sys.now} ${sys.run_id} ${sys.env} 系统注入，只读
${p.*} ${p.region} 管道参数，触发时传入或取默认值
${g.*} ${g.currency} 工作空间全局变量
${nodes.<key>.rows} ${nodes.clean.rows} 上游节点输出指标

bizdate 统一 yyyyMMdd 字符串，日期运算在渲染期完成，避免下游各自解析。

5.3 模板函数

{{ ref('model_name') }}          -- 解析为物理表名，同时产生血缘边与依赖
{{ ref('model','alias') }}       -- 带别名
{{ src('alias','table') }}       -- 引用数据源原始表
{{ macro.partition_range(days=7) }}  -- 生成 dt IN (...) 片段
{{ mask('phone', 'col') }}       -- 按脱敏策略包装列
{{ var('key', 'default') }}

安全边界：模板只允许调用注册过的函数（白名单），禁止任意代码执行；标识符经 QuoteIdent 转义；渲染结果落库审计。

func Render(tpl string, ctx RenderCtx) (sql string, refs []Ref, err error)

type WriteSpec struct {
    Model     string
    Strategy  string   // append|overwrite_partition|merge|scd2|full_refresh|delete_insert
    Partition string
    Keys      []string
    Columns   []string
    Where     string   // delete_insert 用
}

// BuildWriteSQL 根据策略与引擎能力生成写入语句；不支持的策略返回 ErrUnsupported。
func BuildWriteSQL(w WriteSpec, sel string, d dialect.Dialect, cap Capability) (stmt []string, err error)
策略 语义 降级路径
append 直接插入 —
overwrite_partition 删分区 + 插入（同事务/同语句） 依次降级：INSERT OVERWRITE → REPLACE PARTITION（CH）→ DELETE + INSERT（需事务）
merge 按 keys upsert 无 MERGE 用 ON CONFLICT / ON DUPLICATE KEY；都不支持则 DELETE + INSERT
scd2 拉链：关旧链 + 开新链 需事务，无事务引擎拒绝
full_refresh 全量重建（CTAS 或 TRUNCATE+INSERT） —
delete_insert 按 where 删除再插入 —

10. 实现约定（与代码一致，改这里必须同步改代码）

以下几条是实现时做出的收敛决策，优先级高于前文的理想设计。写 YAML 或改代码前先看这一节。

10.1 SQL 变量替换

${name} 会被替换成一个可直接放进 SQL 的完整字面量：

变量值 替换结果
"2026-11-09" '2026-11-09'
100 100
"a'b" 'a''b'（单引号转义）

因此 模板里不要再给变量加引号：写 WHERE dt = ${biz_date}，不要写 WHERE dt = '${biz_date}'（后者会变成 ''2026-11-09''）。

内置变量：biz_date（YYYY-MM-DD）、biz_ts、dt（同 biz_date），其余来自管道 params 与实例参数。未声明的变量在提交时就会被拒绝。

10.2 写入策略

模式 语义 事务
append 直接 INSERT 单语句
overwrite 清空目标表后写入 有 Transactor 则包事务（PG/MySQL），否则逐条（ClickHouse）
upsert ON CONFLICT (unique_key) DO UPDATE 单语句，要求模型声明 unique_key 或 primary_key
overwrite_partition 先 DELETE 当前分区再写入 同 overwrite

原设计的 merge（无冲突键的全表合并）与 replace_partition（ClickHouse 专用）未实现，需要时按 Capability 增加分支。

10.3 实例与租约

queued ──claim(SKIP LOCKED)──▶ running ──成功──▶ success
   ▲                              │
   └── lease 超时 / Recover ──────┴──失败──▶ failed
  • 幂等键：schedule/backfill 为 pipeline_id:biz_date，manual/retry 追加随机后缀以允许重复执行。
  • worker 每 lease/2 续期一次；续期失败（40002）立即取消本地执行，避免双写。
  • 重跑复用 instance.pipeline_ver 指向的版本快照，不看当前 active_version。

10.4 与设计文档不一致的实现选择

项 设计 实现 原因
SQL 访问层 sqlc 生成 pgx 手写 SQL + 参数化 血缘递归、实例分页等动态查询用生成器反而难维护；集中在 repo 包同样可审计
前端 React + 画布 go:embed 内嵌控制台（原生 JS） 单二进制可用、无 npm 构建；画布能力后置，YAML 仍是唯一事实来源
日志 zap log/slog Go 1.21 标准库已满足字段与格式需求，少一个依赖
指标表达式 支持任意 SQL 片段 只允许列名 + 聚合函数白名单 指标是业务口径，不该成为 SQL 注入面
